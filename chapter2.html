3 Pemrosesan Log dengan Spark RDD

# Accumulator dan Parsing Log
acc_rusak = sc.accumulator(0)


def parse_log(line):
    global acc_rusak
    parts = line.split("|")
    if len(parts) == 6:
        try:
            waktu, id_trx, id_user, id_merchant, nominal, status = parts
            return [(waktu, id_trx, id_user, id_merchant, int(nominal), status)]
        except ValueError:
            acc_rusak.add(1)
            return []
    else:
        acc_rusak.add(1)
        return []

rdd_raw = sc.textFile(jalur_log)
rdd_parsed = rdd_raw.flatMap(parse_log).cache()


# Trigger action untuk menghitung accumulator
total_valid = rdd_parsed.count()
print(f"--- (a) Parsing Log ---")

from pyspark.sql.types import StructType, StructField, StringType, LongType, TimestampType
from pyspark.sql import functions as F

# Ubah RDD menjadi DataFrame dengan skema eksplisit
schema_log = StructType([
    StructField("waktu", TimestampType(), True),
    StructField("id_trx", StringType(), True),
    StructField("id_user", StringType(), True),
    StructField("id_merchant", StringType(), True),
    StructField("nominal", LongType(), True),
    StructField("status", StringType(), True)
])



# Mengonversi string timestamp menjadi objek datetime asli Python
rdd_formatted = rdd_parsed.map(
    lambda x: (datetime.strptime(x[0], "%Y-%m-%d %H:%M:%S"), x[1], x[2], x[3], int(x[4]), x[5])
)
df_trx = spark.createDataFrame(rdd_formatted, schema=schema_log).cache()



# DataFrame Merchant dari MongoDB & Broadcast Join dengan Transaksi SUKSES Valid
df_merchant_raw = spark.createDataFrame(merchant)
df_merchant = df_merchant_raw.select(
    F.col("_id").alias("m_id"),
    F.col("kategori"),
    F.col("kota")
)

df_sukses_valid = df_trx.filter(
    (F.col("status") == "SUKSES") & 
    (F.col("nominal") > 0) & 
    (F.col("nominal") < 50_000_000)
)



# Broadcast Join
df_joined = df_sukses_valid.join(
    F.broadcast(df_merchant),
    df_sukses_valid.id_merchant == df_merchant.m_id,
    "inner"
).cache()



# Total nominal per kota per kategori
df_kota_kategori = df_joined.groupBy("kota", "kategori").agg(
    F.sum("nominal").alias("total_nominal"),

5 Ingestion dengan Structured Streaming

import shutil
from pyspark.sql.types import StructType, StructField, StringType, LongType
from pyspark.sql import functions as F

# Skema eksplisit
schema_streaming = StructType([
    StructField("id_trx", StringType(), True),
    StructField("id_merchant", StringType(), True),
    StructField("nominal", LongType(), True),
    StructField("waktu", StringType(), True)
])


dir_parquet = "C:/hadoop/parquet_output"
dir_checkpoint = "C:/hadoop/checkpoint_output"

shutil.rmtree(dir_parquet, ignore_errors=True)
shutil.rmtree(dir_checkpoint, ignore_errors=True)

df_stream = (
    spark.readStream
    .schema(schema_streaming)
    .json(landing)
)

batch_counter = 0


# foreachBatch untuk filter, simpan, dan catat
def process_batch(batch_df, batch_id):
    global batch_counter
    batch_counter += 1
    
    df_valid = batch_df.filter(F.col("nominal").isNotNull())
    df_invalid = batch_df.filter(F.col("nominal").isNull())
    
    print(f"Batch {batch_id} | Valid: {df_valid.count()} event | Tidak Valid (Null): {df_invalid.count()} event")
    df_valid.write.mode("append").parquet(dir_parquet)


# Trigger streaming
query_stream = (
    df_stream.writeStream
    .trigger(availableNow=True)
    .option("checkpointLocation", dir_checkpoint)
    .foreachBatch(process_batch)
    .start()
)

query_stream.awaitTermination()

7 Insight Bisnis & Operasional**

NoteInsight 1

Tingkat Anomali Sistem dan Data Rusak pada Ingestion

  • Temuan: Ditemukan 80 baris transaksi dengan nilai ekstrem Rp250 juta, 150 transaksi nominal negatif (-1), dan 80 baris rusak akibat error jaringan.

  • Bukti: Laporan kualitas data mencatat persis nilai anomali ini pada dataset transaksi, serta accumulator mencatat 80 baris rusak pada pemrosesan log mentah.

  • Tindakan: Mengimplementasikan data validation gate otomatis di tingkat ingestion streaming Kafka/Spark untuk memfilter baris bernilai negatif atau corrupt secara real-time ke Dead Letter Queue (DLQ) sebelum masuk ke database utama.

NoteInsight 2

Pola Beban Transaksi Puncak Harian

  • Temuan: Distribusi transaksi harian merata namun memiliki puncak aktivitas pada jam harian tertentu.

  • Bukti: Grafik histogram menunjukkan distribusi jumlah transaksi sukses per jam dari 0 hingga 23.

  • Tindakan: Mengatur strategi autoscaling pada pod mikroservis pemrosesan pembayaran sebelum jam sibuk harian guna mengurangi tingkat latensi dan kegagalan transaksi (status GAGAL).
